import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Declarative Workflows" icon="document">
    YAML workflows made of typed plugin steps (`StrictParams`), validated fail-fast before anything runs. No loops or spaghetti in YAML.
  </Card>
  <Card title="Agents on a Leash" icon="approve-check-circle">
    The LLM is a step, never the boss. Its output is untrusted, parsed strictly (`FILE:` / exact-match `PATCH:`), bounded by budgets, and followed by compiler & test gates.
  </Card>
  <Card title="Roles, Stacks & Skills as Data" icon="puzzle">
    Define agent roles (`designer`, `developer`, `architect`, `qa`, `reviewer`) with file-path boundaries and dedicated LLMs, plus language stack profiles and `SKILL.md` guidance—all in YAML/Markdown.
  </Card>
  <Card title="Everything Behind Ports" icon="setting">
    Git hosts (`github`, `local-git`, `local-dir`), LLMs (`bedrock`, `openai`, `fixture`), secrets, routers, intakes, and command runners sit behind clean ports. The core never imports a plugin.
  </Card>
</CardGrid>
